package main

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// maxWidth is the widest screenshot we keep as is.
const maxWidth = 320

type Image struct {
	Path  string `json:"path"`
	Width int    `json:"width"`
	Bytes int64  `json:"bytes"`
}

type Store interface {
	Save(img Image) error
}

// Scale returns the ratio that fits the image into maxWidth.
func (img *Image) Scale() float64 {
	if img.Width <= maxWidth {
		return 1.0
	}
	return float64(maxWidth) / float64(img.Width)
}

func collect(dir string, verbose bool) ([]Image, error) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		return nil, fmt.Errorf("read %s: %w", dir, err)
	}
	var images []Image
	for i, entry := range entries {
		name := entry.Name()
		if entry.IsDir() || strings.HasPrefix(name, ".") {
			continue
		}
		info, err := entry.Info()
		if err != nil {
			return nil, err
		}
		images = append(images, Image{Path: filepath.Join(dir, name), Bytes: info.Size()})
		if verbose {
			fmt.Printf("%d: %s\n", i, name)
		}
	}
	return images, nil
}

func main() {
	images, err := collect(".", true)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	fmt.Println(len(images), "images")
}
